**STEP 4 — Add a small fan-out helper inside the hook body** (just before the first useEffect, around line 100):

```javascript
  // Phase 30 — per-annotation CRDT-side fan-out helper. Called after each
  // legacy bulk upsert / delete lands. Runs only when CRDT kill switch is on.
  // Each fabric object is dispatched through dualWriteFabricCommit with
  // skipLegacy: true so the legacy row is not double-written.
  //
  // Highlight bypass at the call site (Pitfall 30-4 layer 1; helper is layer 2).
  // Callout filter at the call site too — callouts ride legacy through Phase 30.
  const fanOutCrdtForAnnotationsByPage = async (annotationsByPageArg, opts) => {
    if (!isCRDTEnabled() || !phase30Ydoc) return;
    const yMapAnnotations = phase30Ydoc.getMap('annotations');
    const originPayload = phase30OriginCtx?.();
    const ctx = phase30UndoCtx;
    for (const [pageNum, page] of Object.entries(annotationsByPageArg || {})) {
      if (!page || !Array.isArray(page.objects)) continue;
      for (const fabricObj of page.objects) {
        const annotation_type = fabricObj?.data?.annotationType || fabricObj?.type;
        // Layer 1 highlight bypass + callout bypass at the call site.
        if (annotation_type === 'highlight' || annotation_type === 'callout') continue;
        if (!NON_HIGHLIGHT_TYPES.includes(annotation_type)) continue;
        try {
          await dualWriteFabricCommit(fabricObj, {
            ...opts,
            ydoc: phase30Ydoc,
            yMapAnnotations,
            originPayload,
            ctx,
            annotation_type,
            skipLegacy: true,  // legacy bulk upsert already fired before this fan-out
          });
        } catch (_err) {
          // Helper enqueues failures internally — no further action needed here.
        }
      }
    }
  };

  const fanOutCrdtForDeletedIds = async (documentId, deletedIds, opts) => {
    if (!isCRDTEnabled() || !phase30Ydoc) return;
    const yMapAnnotations = phase30Ydoc.getMap('annotations');
    const originPayload = phase30OriginCtx?.();
    for (const annoId of (deletedIds || [])) {
      try {
        await dualWriteFabricDelete(documentId, annoId, {
          ...opts,
          ydoc: phase30Ydoc,
          yMapAnnotations,
          originPayload,
          annotation_type: 'fabric',  // generic fabric tag — helper does not filter on this for deletes
          skipLegacy: true,
        });
      } catch (_err) {
        // Helper enqueues failures internally.
      }
    }
  };
```

```javascript
// src/hooks/useTabPendingDualWrite.js
//
// Phase 30 — Per-document subscriber to the dual-write retry queue.
//
// CONTEXT.md `<decisions>` "Document tile signal":
//   When a document has a stuck queue, the document list / sidebar tile shows
//   a small "unsaved changes" icon BEFORE the user opens the document, so they
//   know to check it. Combines with the in-doc banner once they open it.
//
// CONTEXT.md AC-13: per-user (queue is local; CONTEXT.md `<discretion>`
// confirmed by Plan 30-05). Filtered by documentId so each tab pill renders
// independently — drawing 3 different documents and having stuck queues on
// only 1 of them shows the dot on only that 1 tab.
//
// Pattern reference: useDualWriteQueue (Plan 30-05) — same 1Hz polling tick
// with cheap shallow-equal setState gate.

import { useEffect, useState } from 'react';
import { readQueue } from '../lib/collab/crdtDualWriteQueue.js';
import { getAuthSnapshot } from '../supabaseClient.js';

const POLL_INTERVAL_MS = 1_000;

/**
 * @param {string|null|undefined} documentId
 * @returns {boolean} — true iff the dual-write queue has at least one non-quarantined entry for this documentId
 */
export function useTabPendingDualWrite(documentId) {
  const [hasPending, setHasPending] = useState(false);

  useEffect(() => {
    if (!documentId) {
      setHasPending(false);
      return undefined;
    }

    function tick() {
      const auth = getAuthSnapshot ? getAuthSnapshot() : null;
      const userId = auth?.user?.id ?? auth?.userId ?? null;
      if (!userId) {
        setHasPending((prev) => (prev === false ? prev : false));
        return;
      }
      const queue = readQueue(userId) || {};
      let pending = false;
      for (const annoId of Object.keys(queue)) {
        const entry = queue[annoId];
        if (entry?.quarantined) continue;
        // Per-document filter: payload.opts.documentId carries the document
        // through Plan 30-04's enqueue payload shape ({ fabricObj, opts }).
        const entryDocId = entry?.payload?.opts?.documentId ?? entry?.payload?.documentId ?? null;
        if (entryDocId === documentId) {
          pending = true;
          break;
        }
      }
      setHasPending((prev) => (prev === pending ? prev : pending));
    }

    tick();
    const handle = setInterval(tick, POLL_INTERVAL_MS);
    return () => clearInterval(handle);
  }, [documentId]);

  return hasPending;
}

export default useTabPendingDualWrite;
```
