    ```javascript
    import { canModify } from '../lib/collab/permissionScope.js';

    /**
     * Owner-aware post-filter on the marquee hit-test result. Wraps resolveMarqueeHits
     * — collaborator gets only their own marks; owner gets the full hits array reference.
     *
     * @param {Array&lt;number&gt;} hitIndices  // result of resolveMarqueeHits
     * @param {Array&lt;object&gt;} annotations
     * @param {string|null|undefined} viewerId
     * @param {string|null|undefined} documentOwnerId
     * @returns {Array&lt;number&gt;}  owner-mode: same reference; collab-mode: filtered
     */
    export function filterMarqueeHits(hitIndices, annotations, viewerId, documentOwnerId) {
      if (!Array.isArray(hitIndices) || hitIndices.length === 0) return hitIndices;
      if (!viewerId || !documentOwnerId) return hitIndices;  // boot guard
      // Owner-mode: same reference — preserves React equality memoization downstream.
      // Collab-mode: filter to indices whose annotation passes canModify.
      let allOwn = true;
      const filtered = hitIndices.filter((i) =&gt; {
        const a = annotations[i];
        if (!a) return false;
        const ok = canModify(a, viewerId, documentOwnerId);
        if (!ok) allOwn = false;
        return ok;
      });
      // If everything passed AND viewer is owner (canModify always true), return same ref.
      if (allOwn && filtered.length === hitIndices.length) return hitIndices;
      return filtered;
    }
    ```

    UX comment block at top of the new function explaining the owner-mode same-reference optimization.
