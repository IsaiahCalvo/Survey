    import { useState, useEffect, useRef, useCallback } from 'react';

    /**
     * Single-toast undo queue. Enqueueing a new toast replaces any active toast
     * (single-toast queue, not stack — keeps the surface dead simple per CONTEXT.md).
     *
     * Locked enqueue signature (Plan 35-01 frontmatter contract):
     *   enqueue({ kind: 'single' | 'bulk', message, count?, onUndo })
     *
     * Single-delete: 5000ms auto-dismiss. Bulk-delete: 6000ms.
     */
    export function useUndoToast() {
      const [toast, setToast] = useState(null);
      const timerRef = useRef(null);

      const dismiss = useCallback(() =&gt; {
        if (timerRef.current) {
          clearTimeout(timerRef.current);
          timerRef.current = null;
        }
        setToast(null);
      }, []);

      const enqueue = useCallback((input) =&gt; {
        // Locked signature per Plan 35-01: { kind, message, count?, onUndo }
        const kind = input?.kind === 'bulk' ? 'bulk' : 'single';  // default: single
        const next = {
          kind,
          message: input?.message ?? '',
          count: input?.count,
          onUndo: typeof input?.onUndo === 'function' ? input.onUndo : () =&gt; {},
        };
        if (timerRef.current) clearTimeout(timerRef.current);
        const ttl = kind === 'bulk' ? 6000 : 5000;
        setToast(next);
        timerRef.current = setTimeout(() =&gt; {
          setToast(null);
          timerRef.current = null;
        }, ttl);
      }, []);

      useEffect(() =&gt; {
        return () =&gt; {
          if (timerRef.current) clearTimeout(timerRef.current);
        };
      }, []);

      return { toast, enqueue, dismiss };
    }
    ```

      const snapshotObjects = indicesToDelete.map((idx) =&gt; {
        const obj = annotations?.objects?.[idx];
        return obj ? JSON.parse(JSON.stringify(obj)) : null;
      }).filter(Boolean);

      // Build the runDelete closure — this is the existing delete code path.
      const runDelete = () =&gt; {
        const updatedAnnotations = JSON.parse(JSON.stringify(annotations));
        for (const idx of indicesToDelete) {
          updatedAnnotations.objects.splice(idx, 1);
        }
        onSaveAnnotations(updatedAnnotations, {
          source: 'object:modified',
          action: 'delete',
          deletedCount: snapshotObjects.length,  // NEW — App.jsx reads this for single-toast hookup
          deletedSnapshot: snapshotObjects,       // NEW — App.jsx reads this for Undo restore
          deletedPageNumber: pageNumber,          // NEW — restoration target page
          checkpointPolicy: 'normal',
        });
        deselectAll();
      };

      // Phase 35: when the parent provides onRequestBulkDelete, route through
      // the bulk-delete planner. The parent decides modal vs direct-fire.
      // When the prop is absent (e.g., during boot or in test harnesses without
      // App.jsx mounting), fall through to runDelete unconditionally — preserves
      // legacy behavior.
      if (typeof onRequestBulkDelete !== 'function') {
        runDelete();
        return;
      }

      // Build the candidate plan locally so the parent receives a fully-resolved
      // BulkDeletePlan. Use the snapshot objects as the annotations input
      // (ensures consistency between what the modal shows and what runDelete
      // will actually remove).
      const candidateIds = snapshotObjects.map((o) =&gt; o?.id).filter(Boolean);
      // The parent owns viewerId/documentOwnerId — it builds the plan.
      // Pass both candidateIds AND the snapshotObjects so the parent doesn't
      // need to re-resolve them.
      onRequestBulkDelete({
        candidateIds,
        snapshotObjects,
        pageNumber,
        runDelete,
      });
    }, [selectedIds, annotations, onSaveAnnotations, deselectAll, onRequestBulkDelete, pageNumber]);
    ```
