    ```javascript
    /**
     * storageQuotaMonitor — Phase 32 IndexedDB quota polling loop.
     *
     * UX: when local storage usage crosses 80% of quota (browser) or 1 GB
     * (Electron — Pitfall 4 absolute threshold), surface a quiet Tier-1
     * warning. When usage crosses 95% / 4 GB OR a real QuotaExceededError
     * fires, surface a louder Tier-2 banner with "Free up space" action.
     *
     * Source: .planning/phases/32-multi-tab-persistence-hardening/32-RESEARCH.md § Pattern 3, Pitfall 4
     */

    import { attachStorageFailureDetector } from './storageFailureDetector.js';

    // Browser thresholds — percentage of quota.
    export const QUOTA_TIER_PERCENT_BROWSER = Object.freeze({
      tier1: 0.80,
      tier2: 0.95,
    });

    // Electron thresholds — absolute bytes (Pitfall 4: navigator.storage.estimate
    // in Electron reports free DISK space, not the 1/3-of-disk quota — so a
    // percentage threshold never trips reliably).
    export const QUOTA_TIER_ABSOLUTE_ELECTRON = Object.freeze({
      tier1: 1 * 1024 * 1024 * 1024,   // 1 GB
      tier2: 4 * 1024 * 1024 * 1024,   // 4 GB
    });

    const DEFAULT_POLL_MS = 30_000;

    /**
     * @param {object}   [options]
     * @param {object}   [options.windowRef]  — test seam (default: globalThis)
     * @param {number}   [options.pollMs]     — polling cadence (default 30_000)
     * @param {function} [options.onTier]     — ({tier, usage, quota, isElectron}) => void; fires on TIER CHANGE
     * @returns {{ detach: () => void, getQuotaTier: () => 'ok'|'tier1'|'tier2' }}
     */
    export function attachStorageQuotaMonitor(options = {}) {
      const windowRef = options.windowRef
        ?? (typeof globalThis !== 'undefined' ? globalThis : null);
      const pollMs    = options.pollMs ?? DEFAULT_POLL_MS;
      const onTier    = typeof options.onTier === 'function' ? options.onTier : () => {};

      const ua = (windowRef?.navigator?.userAgent ?? '');
      const isElectron = typeof ua === 'string' && ua.includes('Electron');

      let detached = false;
      let lastTier = 'ok';
      let intervalHandle = null;

      const computeTier = (usage, quota) => {
        if (isElectron) {
          // UX: Electron — absolute thresholds (Pitfall 4 defense).
          if (usage >= QUOTA_TIER_ABSOLUTE_ELECTRON.tier2) return 'tier2';
          if (usage >= QUOTA_TIER_ABSOLUTE_ELECTRON.tier1) return 'tier1';
          return 'ok';
        }
        // UX: browser — percentage of quota.
        if (!quota || quota <= 0) return 'ok';
        const pct = usage / quota;
        if (pct >= QUOTA_TIER_PERCENT_BROWSER.tier2) return 'tier2';
        if (pct >= QUOTA_TIER_PERCENT_BROWSER.tier1) return 'tier1';
        return 'ok';
      };

      const setTier = (next, usage = 0, quota = 0) => {
        if (next === lastTier) return;
        lastTier = next;
        try { onTier({ tier: next, usage, quota, isElectron }); } catch { /* swallow */ }
      };

      const pollOnce = async () => {
        if (detached) return;
        const estimate = windowRef?.navigator?.storage?.estimate;
        if (typeof estimate !== 'function') return;   // SSR / unsupported
        try {
          const { usage = 0, quota = 0 } = await estimate.call(windowRef.navigator.storage);
          if (detached) return;
          setTier(computeTier(usage, quota), usage, quota);
        } catch {
          /* swallow — estimate() can throw in cross-origin iframes */
        }
      };

      // Start interval + run an immediate poll.
      pollOnce();
      intervalHandle = setInterval(pollOnce, pollMs);

      // Subscribe to storageFailureDetector — QuotaExceededError forces tier2.
      const detector = attachStorageFailureDetector({
        windowRef,
        onState: (state) => {
          if (detached) return;
          if (state && state.code === 'quota_exceeded') {
            setTier('tier2', 0, 0);
          }
        },
      });

      return {
        detach() {
          detached = true;
          if (intervalHandle) {
            try { clearInterval(intervalHandle); } catch { /* swallow */ }
            intervalHandle = null;
          }
          try { detector?.detach?.(); } catch { /* swallow */ }
        },
        getQuotaTier: () => lastTier,
      };
    }
    ```

    Critical guardrails:
    - DO NOT call `persistence.clearData()` or any IDB-destructive API. This file is read-only against IDB state.
    - DO NOT add a fourth tier ("tier3", "critical", etc.). Locked to ok/tier1/tier2 by CONTEXT.md.
    - The `windowRef` seam is required for the unit tests — production callers pass nothing and the default `globalThis` is used.
    - `attachStorageFailureDetector` is the existing Phase 27 module — its `{ windowRef }` option is already supported per Plan 27-04. Reuse, do not duplicate.
    - `onTier` MUST only fire on TIER CHANGE — repeated polls at the same tier are silent. Tests assert this.
