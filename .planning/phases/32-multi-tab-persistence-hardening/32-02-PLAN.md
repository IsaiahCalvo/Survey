<objective>
Ship the periodic Y.Doc snapshot compaction job (`yDocCompaction.js`) and the atomic Postgres RPC (`compact_yjs_doc`) that defends Pitfall 1 (Non-Atomic Snapshot/Prune). The job is invoked by the leader tab when 100 Y.Doc updates have accumulated since the last snapshot — writes a single `doc_yjs_state` snapshot row + prunes `doc_yjs_updates` rows below `through_seq` in ONE Postgres transaction.

Purpose: cold-loads stay fast as the document accumulates edits — without server-side compaction, new sessions replay tens of thousands of `doc_yjs_updates` rows. With it, cold-load is one `SELECT state FROM doc_yjs_state` + a short tail of recent updates.

Output: 1 new JS module (`src/lib/collab/yDocCompaction.js`) + 1 new Supabase migration (`supabase/migrations/20260520000000_phase32_compaction_rpc.sql`). Plan 32-01's 4 yDocCompaction tests + 1 compactionRpcAtomic test flip skip → green.

    ```sql
    -- Migration: Phase 32 — Server-side atomic Y.Doc snapshot write + prune.
    -- Defends Pitfall 1 (Snapshot/Prune Non-Atomic).

    CREATE OR REPLACE FUNCTION compact_yjs_doc(
      p_document_id UUID,
      p_state BYTEA,
      p_state_vector BYTEA
    )
    RETURNS TABLE(through_seq BIGINT, bytes BIGINT)
    LANGUAGE plpgsql
    SECURITY DEFINER

    AS $$

      v_through_seq BIGINT;
    BEGIN
      IF NOT public.user_can_access_document(auth.uid(), p_document_id) THEN
        RAISE EXCEPTION 'permission_denied' USING ERRCODE = '42501';
      END IF;

      SELECT COALESCE(MAX(seq), 0) INTO v_through_seq
        FROM public.doc_yjs_updates
        WHERE document_id = p_document_id;

      INSERT INTO public.doc_yjs_state (document_id, state, state_vector, through_seq, encoding_version, updated_at)
        VALUES (p_document_id, p_state, p_state_vector, v_through_seq, 1, NOW())
      ON CONFLICT (document_id) DO UPDATE SET
        state = EXCLUDED.state,
        state_vector = EXCLUDED.state_vector,
        through_seq = EXCLUDED.through_seq,
        updated_at = EXCLUDED.updated_at;

      DELETE FROM public.doc_yjs_updates
        WHERE document_id = p_document_id
          AND seq <= v_through_seq;

      RETURN QUERY SELECT v_through_seq AS through_seq, OCTET_LENGTH(p_state)::BIGINT AS bytes;
    END;
    $$;
    ```

    ```javascript
    export function attachCompaction(ydoc, documentId, options = {}) {
      if (!ydoc || !documentId) {
        throw new Error('[yDocCompaction] ydoc + documentId required');
      }
      const threshold   = options.threshold ?? DEFAULT_THRESHOLD;
      const supabase    = options.supabase ?? defaultSupabase;   // test seam
      const onCompacted = options.onCompacted ?? (() => {});
      const onError     = options.onError ?? (() => {});

      let updatesSinceLastSnapshot = 0;
      let scheduled = false;
      let detached = false;

      const runCompaction = async () => {
        if (detached) return;
        scheduled = false;
        try {
          const state = Y.encodeStateAsUpdate(ydoc);
          const stateVector = Y.encodeStateVector(ydoc);
          // UX: atomic snapshot + prune via single Postgres function (Pitfall 1).
          const { data, error } = await supabase.rpc('compact_yjs_doc', {
            p_document_id: documentId,
            p_state: state,
            p_state_vector: stateVector,
          });
          if (error) throw error;
          updatesSinceLastSnapshot = 0;
          const row = Array.isArray(data) ? data[0] : data;
          onCompacted({
            throughSeq: row?.through_seq ?? null,
            bytes: row?.bytes ?? state.byteLength,
          });
        } catch (err) {
          // NOTE: do NOT reset updatesSinceLastSnapshot on error — the next
          // threshold trip will retry. Pitfall-3 starvation guard is the
          // 5s requestIdleCallback timeout, not error-reset.
          onError(err);
        }
      };

      const schedule = () => {
        if (scheduled || detached) return;
        scheduled = true;
        if (typeof requestIdleCallback === 'function') {
          requestIdleCallback(runCompaction, { timeout: IDLE_TIMEOUT_MS });
        } else {
          setTimeout(runCompaction, 200);   // Safari / SSR fallback
        }
      };

      const onUpdate = () => {
        if (detached) return;
        updatesSinceLastSnapshot += 1;
        if (updatesSinceLastSnapshot >= threshold) schedule();
      };

      ydoc.on('update', onUpdate);
      return {
        detach() {
          detached = true;
          try { ydoc.off('update', onUpdate); } catch { /* swallow */ }
        },
        _forceCompactForTest: runCompaction,
      };
    }
    ```
