    ```javascript
    /**
     * multiTabSync — Phase 32 per-docId BroadcastChannel awareness coordinator.
     *
     * UX: when a user opens the same document in two browser windows of the same
     * origin (the only path NOT blocked by the in-window same-PDF guard), edits
     * in window A surface awareness signals (active writer, cursor focus) to
     * window B within ~1 second.
     *
     * This file ONLY handles AWARENESS messages on a DEDICATED channel named
     * `y-doc-aware-${documentId}`. Y.Doc updates remain on Phase 27's existing
     * `y-doc-bc-${documentId}` channel inside ydocLifecycle.js. Two separate
     * channels keep the echo-guard contracts independent — duplicating Y.Doc
     * updates here would double-apply and re-amplify across tabs (Pitfall 2).
     *
     * The visible "who is editing right now" UI is Phase 33's presence pill —
     * Phase 32 only ships the data plumbing.
     *
     * Source: .planning/phases/32-multi-tab-persistence-hardening/32-RESEARCH.md § Pattern 2
     */

    import * as awarenessProtocol from 'y-protocols/awareness';

    export const REMOTE_AWARE_ORIGIN = Object.freeze({ source: 'remote-aware-bc' });

    /**
     * @param {import('yjs').Doc} ydoc            — local Y.Doc; clientID is used as the self-echo key
     * @param {object|null}        awareness       — y-protocols Awareness instance (or null in Phase 32 if Phase 33 hasn't shipped yet)
     * @param {string}             documentId      — per-docId channel scoping
     * @param {object}             [options]
     * @param {typeof BroadcastChannel} [options.BroadcastChannelCtor] — test seam (default: globalThis.BroadcastChannel)
     * @returns {{ detach: () => void }}
     */
    export function attachMultiTabSync(ydoc, awareness, documentId, options = {}) {
      if (!ydoc || !documentId) {
        throw new Error('[multiTabSync] ydoc + documentId required');
      }
      const BroadcastChannelCtor = options.BroadcastChannelCtor
        ?? (typeof BroadcastChannel !== 'undefined' ? BroadcastChannel : null);

      // SSR / unsupported browser fallback: return a no-op detach so callers do
      // not need to branch. Phase 27 ydocLifecycle.js uses the same pattern.
      if (!BroadcastChannelCtor) {
        return { detach: () => {} };
      }

      const channelName = `y-doc-aware-${documentId}`;
      const bc = new BroadcastChannelCtor(channelName);
      let detached = false;

      // ------------------------------------------------------------------
      // OUTBOUND: awareness 'update' events broadcast changedClients' state.
      // Skip when the just-applied awareness change came FROM another tab —
      // that's an echo (Pitfall 2 defense, mirror of REMOTE_BC_ORIGIN check
      // in ydocLifecycle.js).
      // ------------------------------------------------------------------
      const onAwarenessUpdate = ({ added, updated, removed }, transactionOrigin) => {
        if (detached || !awareness) return;
        if (transactionOrigin === REMOTE_AWARE_ORIGIN) return;   // do not re-amplify
        const changedClients = added.concat(updated).concat(removed);
        if (changedClients.length === 0) return;
        const payload = awarenessProtocol.encodeAwarenessUpdate(awareness, changedClients);
        try {
          bc.postMessage({
            type: 'awareness',
            payload,
            originClientId: ydoc.clientID,
          });
        } catch {
          /* swallow — BC postMessage can throw if channel is closed mid-detach */
        }
      };
      if (awareness && typeof awareness.on === 'function') {
        awareness.on('update', onAwarenessUpdate);
      }

      // ------------------------------------------------------------------
      // INBOUND: messages from other tabs of the same origin + same docId.
      // Drop self-echo by clientID; otherwise apply with REMOTE_AWARE_ORIGIN
      // so the outbound listener above skips re-broadcasting.
      // ------------------------------------------------------------------
      bc.onmessage = (ev) => {
        if (detached) return;
        if (!ev || !ev.data) return;
        if (ev.data.originClientId === ydoc.clientID) return;   // self-echo guard
        if (ev.data.type === 'awareness' && awareness && ev.data.payload) {
          try {
            const u8 = ev.data.payload instanceof Uint8Array
              ? ev.data.payload
              : new Uint8Array(ev.data.payload);
            awarenessProtocol.applyAwarenessUpdate(awareness, u8, REMOTE_AWARE_ORIGIN);
          } catch {
            /* swallow malformed payloads — channel is best-effort */
          }
        }
      };

      return {
        detach() {
          detached = true;
          try {
            if (awareness && typeof awareness.off === 'function') {
              awareness.off('update', onAwarenessUpdate);
            }
          } catch { /* swallow */ }
          try { bc.close(); } catch { /* swallow */ }
        },
      };
    }
    ```

    Critical guardrails:
    - File MUST NOT `bc.postMessage({ type: 'sync' ...})` or any non-awareness message type. Phase 27's existing `y-doc-bc-${docId}` channel owns Y.Doc update fan-out — duplicating it here re-amplifies the loop.
    - File MUST NOT import anything from `src/lib/collab/ydocLifecycle.js`. The two coordinators run parallel; coupling them creates a circular-attach hazard.
    - The string `y-doc-aware-` is hard-coded — Plan 32-06's YDocProvider integration relies on this name to inspect channel state in dev tools.
    - The export name `REMOTE_AWARE_ORIGIN` is locked — Plan 33 will read this same sentinel from Phase 32 to discriminate the awareness publisher's outbound (don't re-publish to the realtime wire on remote-aware-bc inbound).
