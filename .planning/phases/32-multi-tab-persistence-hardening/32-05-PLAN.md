    ```javascript
    /**
     * getSyncStatusViewModel — single source of truth for SyncStatusChip's
     * three locked states. Pure function — no React, no DOM access.
     *
     * CONTEXT.md (Phase 32) locked these strings character-for-character:
     *   - 'Offline — N changes queued'   (em-dash U+2014; N==1 → 'change')
     *   - 'Syncing N changes…'           (ellipsis U+2026; N==1 → 'change')
     *   - 'Syncing…'                     (when queueSize is 0)
     *   - 'Up to date'                   (idle)
     *
     * The quotaTier 4th arg threads the Phase 32 storage-quota signal through
     * to the chip without changing the locked label string. The chip reads
     * quotaTier separately to render the Tier-1 amber ring + tooltip suffix.
     */

    export function getSyncStatusViewModel(status, queueSize = 0, manualSyncing = false, quotaTier = 'ok') {
      const stage = status?.stage || 'idle';

      if (manualSyncing) {
        return { state: 'syncing', label: 'Syncing now...', quotaTier };
      }
      if (stage === 'error') {
        return { state: 'offline', label: 'Sync error', quotaTier };
      }

      // OFFLINE state: queue is non-empty OR explicitly queued stage.
      if (queueSize > 0 || stage === 'queued') {
        if (queueSize > 0) {
          // CONTEXT.md locked copy: 'Offline — N changes queued' (em-dash U+2014).
          return {
            state: 'offline',
            label: `Offline — ${queueSize} ${queueSize === 1 ? 'change' : 'changes'} queued`,
            quotaTier,
          };
        }
        // queueSize=0 with stage='queued' is the legacy "saved locally, no count" fallback.
        return { state: 'offline', label: 'Saved locally', quotaTier };
      }

      // SYNCING state: any in-flight or hydrating sync work.
      if (stage === 'pending' || stage === 'syncing' || stage === 'hydrating' || stage === 'migrating') {
        if (queueSize > 0) {
          // CONTEXT.md locked copy: 'Syncing N changes…' (ellipsis U+2026).
          return {
            state: 'syncing',
            label: `Syncing ${queueSize} ${queueSize === 1 ? 'change' : 'changes'}…`,
            quotaTier,
          };
        }
        // No count yet — initial hydration etc.
        return { state: 'syncing', label: 'Syncing…', quotaTier };
      }

      // UP TO DATE state.
      // CONTEXT.md locked copy: 'Up to date'.
      return { state: 'synced', label: 'Up to date', quotaTier };
    }
    ```

    Notes:
    - Em-dash (`—`) and ellipsis (`…`) are real Unicode characters in the source file — NOT `—` HTML entities, NOT three ASCII periods. Plan 32-01 tests assert with the literal characters.

    - The function signature gains a 4th arg `quotaTier='ok'` — backward-compatible. The chip will pass it; older call-sites without the arg get `'ok'` default and behave as before.
    - Comments may use words like `IndexedDB` / `Y.Doc` / etc. — but the RETURNED LABEL strings (user-visible) only use plain English per `feedback_plain_english` user preference.
