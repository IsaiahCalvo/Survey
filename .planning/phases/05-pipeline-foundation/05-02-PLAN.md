   ```javascript
         name: 'chromium',
         use: { channel: 'chromium' },
       },
     ],
   });
   ```

   Key points:
   - `timeout: 120_000` because Syncfusion cold start can take 15-30 seconds
   - `webServer.reuseExistingServer: true` so manually-started dev servers are reused (faster iteration)

3. **Create `debug/lib/session.mjs`** with session folder creation and manifest finalization:

   Exports:
   - `createSession(scenarioName, baseDir)` -- creates timestamped folder, captures git SHA, returns `{ sessionDir, manifest, folderName }`
     - Timestamp format: `YYYYMMDDTHHMMSS` (ISO 8601 compact, sortable) e.g., `20260312T143022`
     - Folder: `debug/debug-sessions/<timestamp>_<scenario>/`
     - Git SHA: via `execSync('git rev-parse HEAD', { encoding: 'utf-8' }).trim()`
     - Initial manifest: `{ scenario, gitSha, startTime: ISO string, endTime: null, result: null, artifacts: [] }`

   - `finalizeSession(sessionDir, manifest, result, artifacts)` -- writes manifest.json to session folder
     - Sets `endTime` to current ISO string
     - Sets `result` to `'pass'` or `'fail'`
     - Sets `artifacts` array (each: `{ type, path, description }`)
     - Writes `JSON.stringify(manifest, null, 2)` to `<sessionDir>/manifest.json`

   - `getSessionBaseDir()` -- returns the absolute path to `debug/debug-sessions/` relative to the config file location
     - Use `import.meta.url` to resolve relative to the lib file: `path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..', 'debug-sessions')`

   Implementation details:
   - Use `node:fs` (mkdirSync, writeFileSync), `node:path` (join, resolve, dirname), `node:child_process` (execSync)
   - Use `import { fileURLToPath } from 'node:url'` for import.meta.url resolution
   - Create `debug/debug-sessions/` directory with `{ recursive: true }` if it doesn't exist
   - Handle `git rev-parse HEAD` failure gracefully (return 'unknown' if not in a git repo)

**Test: "smoke test - loads PDF, captures canvas content, creates session"**

Steps:
1. **Create session folder** at test start:
   ```javascript
   import { createSession, finalizeSession, getSessionBaseDir } from '../lib/session.mjs';
   const baseDir = getSessionBaseDir();
   const { sessionDir, manifest } = createSession('smoke', baseDir);
   ```

2. **Navigate to dev test route:**
   ```javascript
   await page.goto('/?testPdf=Package%202%20-%20Rev%204%20--%20IC.pdf');
   ```
   (baseURL from config provides the host)

3. **Wait for PDF viewer to load:**
   - Wait for the Syncfusion viewer container: `await page.locator('.e-pv-viewer-container').waitFor({ state: 'visible', timeout: 60_000 })`
   - Wait for initial page render: `await page.waitForTimeout(5000)` (conservative wait for Syncfusion + PDF.js initialization)

4. **Navigate to page 6** (the page with annotations):
   ```javascript
   const pageInput = page.getByRole('textbox', { name: 'Current page' });
   await pageInput.click();
   await pageInput.fill('6');
   await pageInput.press('Enter');
   await page.waitForTimeout(5000);  // Wait for page change + Fabric.js render
   ```

5. **Wait for Fabric.js canvas content:**
   - Wait for `.canvas-container` elements: `await page.locator('.canvas-container').first().waitFor({ state: 'visible', timeout: 30_000 })`
   - Verify canvas has actual content (not blank) by evaluating pixel data:
   ```javascript
   const hasCanvasContent = await page.evaluate(() => {
     const containers = document.querySelectorAll('.canvas-container');
     if (containers.length === 0) return false;

     // Check each canvas container for non-blank content
     for (const container of containers) {
       const canvases = container.querySelectorAll('canvas');
       for (const canvas of canvases) {
         const ctx = canvas.getContext('2d');
         if (!ctx) continue;
         try {
           const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
           const data = imageData.data;
           let nonBlankPixels = 0;
           // Sample every 100th pixel for speed
           for (let i = 0; i < data.length; i += 400) {
             const r = data[i], g = data[i+1], b = data[i+2], a = data[i+3];
             if (a > 0 && !(r === 255 && g === 255 && b === 255)) {
               nonBlankPixels++;
             }
           }
           if (nonBlankPixels > 10) return true;  // Found canvas with content
         } catch (e) {
           // getImageData can fail on tainted canvases -- skip
           continue;
         }
       }
     }
     return false;
   });
   ```
   - This is the PHASE 5 GATE. If `hasCanvasContent` is false, the test fails with a descriptive message: "PHASE 5 GATE FAILED: Fabric.js canvas content not captured in screenshot. Canvas containers found but content is blank."

6. **Take screenshot of page 6:**
   ```javascript
   const screenshotPath = path.join(sessionDir, 'page6-annotated.png');
   await page.screenshot({ path: screenshotPath, fullPage: false });
   ```

7. **Assertions:**
   - `expect(hasCanvasContent).toBe(true)` -- canvas has non-blank pixels (Phase 5 gate)
   - Verify screenshot file exists and has non-zero size
   - Verify the number of `.canvas-container` elements is >= 1

8. **Finalize session:**
   ```javascript
   const artifacts = [
     { type: 'screenshot', path: 'page6-annotated.png', description: 'Page 6 with Fabric.js annotations' }
   ];
   const result = hasCanvasContent ? 'pass' : 'fail';
   finalizeSession(sessionDir, manifest, result, artifacts);
   ```

9. **Verify manifest:**
   ```javascript
   const manifestData = JSON.parse(fs.readFileSync(path.join(sessionDir, 'manifest.json'), 'utf-8'));
   expect(manifestData.scenario).toBe('smoke');
   expect(manifestData.gitSha).toMatch(/^[0-9a-f]{40}$/);
   expect(manifestData.startTime).toBeTruthy();
   expect(manifestData.endTime).toBeTruthy();
   expect(manifestData.result).toBe('pass');
   expect(manifestData.artifacts).toHaveLength(1);
   expect(manifestData.artifacts[0].type).toBe('screenshot');
   ```

**Import requirements at the top:**
```javascript
import { test, expect } from '@playwright/test';
import { createSession, finalizeSession, getSessionBaseDir } from '../lib/session.mjs';
import { mkdirSync, readFileSync, statSync } from 'node:fs';
import path from 'node:path';
```

**Error handling:**
- If the PDF fails to load (no viewer container after 60s), fail with "PDF viewer did not load -- check dev test route"
- If page navigation fails (page input not found), fail with "Page navigation input not found -- Syncfusion toolbar may not have rendered"
- If canvas containers are found but all blank, fail with the Phase 5 gate message
- Use `test.afterEach` to finalize the session even if the test fails (so we get a manifest with result='fail' and any partial artifacts)

**Key constraints:**
- Do NOT use `slowMo` (user decision: normal speed preserves timing fidelity)
- The test should complete in under 90 seconds (typical: 30-60s including Vite cold start)
- Conservative timeouts throughout -- Syncfusion + Fabric.js rendering is slow
