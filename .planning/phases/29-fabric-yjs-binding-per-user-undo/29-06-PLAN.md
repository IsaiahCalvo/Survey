    ```jsx
    // src/components/collab/CollaboratorOutlineOverlay.jsx
    // Phase 29 — Sibling SVG overlay drawing per-user-color outlines around annotations
    // that remote collaborators currently have open in their edit canvas.
    //
    // UI-SPEC §2 visual contract: 2px solid stroke, 4px outset offset, opacity 0.7,
    // border-radius 4px, pointer-events none, 160ms fade in/out animation.
    //
    // Why sibling overlay (NOT modifying SVGAnnotationLayer):
    //   SVGAnnotationLayer is Always-Protected (CLAUDE.md). The outline is purely additive
    //   chrome that overlays on top — implemented as a sibling positioned absolute over
    //   the same per-page coordinate space.
    //
    // Why one outline color slot per remote user:
    //   UI-SPEC + CONTEXT.md "Awareness signal" — stable per-user color assigned by server.
    //   Phase 33 reuses the same colors for the cursor pill / presence list, so the user
    //   has a continuous mental model when more awareness UI ships.

    import React, { useMemo } from 'react';
    import './CollaboratorOutlineOverlay.css';

    /**
     * @param {object} props
     * @param {number} props.pageNumber - Which PDF page this overlay is rendering for.
     * @param {{ width: number, height: number }} props.pageSize - Page dimensions in viewer units.
     * @param {Array<{ annoId: string, bbox: { left, top, width, height }, userId: string, colorSlot: number }>} props.editors
     *   - One entry per remote-edit-active annotation. colorSlot is 1..6 mapping to --user-color-N CSS var.
     */
    export function CollaboratorOutlineOverlay({ pageNumber, pageSize, editors = [] }) {
      const rects = useMemo(() => {
        return editors.map((e) => {
          // 4px outset offset per UI-SPEC.
          const offset = 4;
          return {
            key: `${e.annoId}-${e.userId}`,
            x: e.bbox.left - offset,
            y: e.bbox.top - offset,
            width: e.bbox.width + offset * 2,
            height: e.bbox.height + offset * 2,
            colorVar: `var(--user-color-${((e.colorSlot - 1) % 6) + 1}, #4A90E2)`,
          };
        });
      }, [editors]);

      if (!pageSize || rects.length === 0) return null;

      return (
        <svg
          className="collaborator-outline-overlay"
          width={pageSize.width}
          height={pageSize.height}
          viewBox={`0 0 ${pageSize.width} ${pageSize.height}`}
          aria-hidden="true"  /* Outline is informational decoration; not announced */
        >
          {rects.map((r) => (
            <rect
              key={r.key}
              className="collaborator-outline-overlay__rect"
              x={r.x}
              y={r.y}
              width={r.width}
              height={r.height}
              style={{ stroke: r.colorVar }}
            />
          ))}
        </svg>
      );
    }

    export default CollaboratorOutlineOverlay;
    ```

    **Step C — Create `src/hooks/useRemoteEditors.js`**:

    ```js
    // src/hooks/useRemoteEditors.js
    // Phase 29 — Read Yjs awareness state for remote editors of annotations on the current page.
    //
    // If the Phase 28 transport provider exposes awareness via globalThis.__crdtTransportProvider
    // OR via the YDocContext, read from there. Graceful fallback: return empty Map when
    // awareness is not yet wired (Phase 28 may have partial awareness coverage).

    import { useState, useEffect } from 'react';
    import { useYDoc } from './useYDoc.js';

    /** @returns {Map<string, { userId: string, colorSlot: number, name: string }>} */
    export function useRemoteEditors() {
      const { ydoc } = useYDoc();
      const [editors, setEditors] = useState(() => new Map());

      useEffect(() => {
        if (!ydoc) {
          setEditors(new Map());
          return;
        }
        // Awareness state lookup: Phase 28's SupabaseYjsProvider attaches awareness
        // to the Y.Doc — typically as ydoc.awareness or via a sibling provider object.
        // Defensive: try both common patterns.
        const awareness = ydoc.awareness ?? globalThis.__crdtAwareness ?? null;
        if (!awareness) {
          setEditors(new Map());
          return;
        }

        const update = () => {
          const next = new Map();
          const states = awareness.getStates?.();
          if (!states) {
            setEditors(next);
            return;
          }
          for (const [clientId, state] of states.entries()) {
            if (clientId === awareness.clientID) continue;  // skip self
            const editingId = state?.editingAnnotationId;
            if (!editingId) continue;
            next.set(editingId, {
              userId: state?.user?.id ?? `client-${clientId}`,
              colorSlot: state?.user?.colorSlot ?? 1,
              name: state?.user?.name ?? 'collaborator',
            });
          }
          setEditors(next);
        };

        update();
        awareness.on?.('change', update);
        return () => awareness.off?.('change', update);
      }, [ydoc]);

      return editors;
    }

    export default useRemoteEditors;
    ```

    ```jsx
    // Phase 29 — Remote-delete toast queue (Plan 29-06 Task 3).
    //
    // Surfaces ONLY when local user is interacting with the deleted annotation (selected /
    // dragging / scaling / edit-canvas open / context menu open). Plan 29-05 publishes
    // window.__phase29InteractionState with these fields; we read it here.
    //
    // Sticky (no auto-dismiss): UI-SPEC §1 contract.
    const [toasts, setToasts] = useState([]);

    useEffect(() => {
      if (!ydoc || !undoState?.undoCtx) return;
      const yMap = ydoc.getMap('annotations');
      const handler = (event, transaction) => {
        // Skip our own writes.
        if (transaction.origin?.source === 'local-fabric') return;
        if (transaction.origin?.source === 'local-undo') return;
        if (transaction.origin?.source === 'local-redo') return;

        const interactionState = (typeof window !== 'undefined' && window.__phase29InteractionState) || {};

        // event.changes.keys is a Map<key, { action: 'add' | 'update' | 'delete', oldValue }>
        event.changes?.keys?.forEach((change, annoId) => {
          if (change.action !== 'delete') return;
          // Was local user interacting with annoId at this moment?
          const interacting = (
            interactionState.selectedId === annoId
            || interactionState.draggingId === annoId
            || interactionState.scalingId === annoId
            || interactionState.editCanvasId === annoId
            || interactionState.contextMenuId === annoId
          );
          if (!interacting) return;

          // Recover oldValue snapshot for restore.
          const snapshotJSON = change.oldValue?.toJSON?.();
          if (!snapshotJSON) return;

          const collaboratorName = transaction.origin?.userName ?? null;

          setToasts((prev) => [
            ...prev,
            {
              id: `toast-${annoId}-${Date.now()}`,
              code: 'annotation_remote_deleted',
              collaboratorName,
              annoId,
              snapshotJSON,
            },
          ]);
        });
      };
      yMap.observe(handler);
      return () => yMap.unobserve(handler);
    }, [ydoc, undoState?.undoCtx]);

    const handleRestore = useCallback((toast) => {
      // Warning 4 resolution: getLocalFabricOrigin imported SYNCHRONOUSLY at top of file.
      // No dynamic await import.
      if (!ydoc || !undoState?.undoCtx || !toast?.snapshotJSON) return;
      const yMap = ydoc.getMap('annotations');
      const ctx = undoState.undoCtx;
      const originPayload = getLocalFabricOrigin(ctx);

      ydoc.transact(() => {
        // Skip if Y.Map already has this annoId (race: collaborator un-deleted faster).
        if (yMap.get(toast.annoId)) return;
        const annoYMap = new Y.Map();
        const fabricYMap = new Y.Map();
        const metaYMap = new Y.Map();
        yMap.set(toast.annoId, annoYMap);
        annoYMap.set('id', toast.annoId);
        annoYMap.set('type', toast.snapshotJSON.type);
        annoYMap.set('pageNumber', toast.snapshotJSON.pageNumber);
        annoYMap.set('fabric', fabricYMap);
        annoYMap.set('meta', metaYMap);
        // Preserve original CREATE-meta from snapshot (UNDO-03 contract).
        // Why direct write (not applyFabricCommit): applyFabricCommit's CREATE branch sets
        // meta.authorId/deviceId/createdAt from ctx (the restoring user). UNDO-03 says
        // original CREATE-meta must survive — so we manually write the snapshot's meta block.
        const oldMeta = toast.snapshotJSON.meta ?? {};
        metaYMap.set('authorId', oldMeta.authorId);
        metaYMap.set('deviceId', oldMeta.deviceId);
        metaYMap.set('createdAt', oldMeta.createdAt);
        // Update timestamps reflect the restore operation.
        metaYMap.set('updatedAt', Date.now());
        metaYMap.set('lastEditorId', ctx.userId);
        metaYMap.set('restoredBy', ctx.userId);
        // Restore fabric properties.
        const fabricSnapshot = toast.snapshotJSON.fabric ?? {};
        Object.keys(fabricSnapshot).forEach((k) => {
          fabricYMap.set(k, fabricSnapshot[k]);
        });
      }, originPayload);

      // Dismiss this toast.
      setToasts((prev) => prev.filter((t) => t.id !== toast.id));
    }, [ydoc, undoState?.undoCtx]);

    const handleDismissToast = useCallback((toastId) => {
      setToasts((prev) => prev.filter((t) => t.id !== toastId));
    }, []);
    ```
