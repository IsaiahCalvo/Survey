<objective>
Simplify PAL's scale useEffect by removing its independent 300ms settle timer, removing dead props (onScaleApplied, presentationApiRegistry, isHidden), removing the wrapper CSS transform logic, and ensuring the post-zoom redraw sequence follows the strict order: redraw canvas -> remove CSS transform -> restore pointer events.

Purpose: PAL becomes a "slave" to App.jsx's zoom state. The scale prop change (frozen -> live) IS the settle signal. This eliminates an entire class of race conditions from independent timers.

Output: Modified `src/PageAnnotationLayer.jsx` (scale useEffect simplified, 3 props removed, 5 onScaleApplied call sites removed) and `src/App.jsx` (onScaleApplied prop removed from PAL render, handlePALScaleApplied cleaned up).

**Part C: Simplify the scale useEffect (PageAnnotationLayer.jsx lines 7788-8028)**

This is the core change. The new useEffect structure:

```javascript
useEffect(() => {
  // REMOVED: if (isHidden) return;
  if (!fabricRef.current || !width || !height) return;

  const canvas = fabricRef.current;
  const currentZoom = canvas.getZoom();
  const scaleJump = Math.abs(scale - currentZoom);

  // Interaction deferral (KEEP -- separate concern from zoom)
  if (isInteracting) {
    pendingScaleRef.current = scale;
    return;
  }

  // Enter zoom mode on any zoom signal or large scale jump.
  // Once latched, only the settle path below clears it.
  if (isZooming || scaleJump > 0.01) {
    inZoomModeRef.current = true;
  }

  // -- Deferred resize path (zoom mode active) --
  if (inZoomModeRef.current) {
    const prevPending = pendingScaleRef.current;
    pendingScaleRef.current = scale;

    const scaleActuallyChanged = prevPending === null || Math.abs(scale - prevPending) > 0.001;
    if (!scaleActuallyChanged) return;

    // Cancel any pending resize rAF (zoom resumed before it fired)
    cancelPendingPaintCommit();
    if (scaleUpdateFrameRef.current) {
      cancelAnimationFrame(scaleUpdateFrameRef.current);
      scaleUpdateFrameRef.current = null;
    }

    // Cancel any pending viewport observer or deferred timer
    if (viewportObserverRef.current) {
      viewportObserverRef.current.disconnect();
      viewportObserverRef.current = null;
    }
    if (deferredZoomScaleRef.current?._timerId) {
      clearTimeout(deferredZoomScaleRef.current._timerId);
    }
    deferredZoomScaleRef.current = null;

    // REMOVED: 300ms zoomSettleTimerRef timer.
    // Instead: if isZooming is still true, just store pending and return.
    // If isZooming is false, this scale change IS the settle signal
    // (it fires when App.jsx's overlay settle timer causes layerScale
    // to switch from frozen to live value).
    if (isZoomingRef.current) {
      return;
    }

    // Zoom settled! Execute tiered redraw.
    const finalScale = pendingScaleRef.current ?? scale;
    pendingScaleRef.current = null;
    const tw = Math.floor(width * finalScale);
    const th = Math.floor(height * finalScale);

    // Clear zoom latch before enqueueing.
    inZoomModeRef.current = false;

    const wrapperEl = canvas.wrapperEl;

    // Classify page priority
    let isCenterPage = false;
    let isVisible = false;
    if (wrapperEl) {
      const rect = wrapperEl.getBoundingClientRect();
      isVisible = rect.bottom > 0 && rect.top < window.innerHeight;
      if (isVisible) {
        const vcY = window.innerHeight / 2;
        isCenterPage = rect.top <= vcY && rect.bottom >= vcY;
      }
    }

    // REMOVED: wrapper CSS transform for non-center pages (old lines 7891-7894)
    // In the new design, App.jsx removes overlay div transforms globally on settle.
    // Non-center pages show CSS-scaled (blurry) overlay until their delayed Fabric render fires.

    // Helper: perform Fabric.js resize + render
    const doFabricRender = () => {
      if (inZoomModeRef.current) return;
      const fc = fabricRef.current;
      if (!fc) return;
      deferredZoomScaleRef.current = null;
      // REMOVED: wrapper CSS transform cleanup (old lines 7903-7906)
      // Transforms are managed by App.jsx on the overlay div, not by PAL on the wrapper.
      fc.setWidth(tw);
      fc.setHeight(th);
      fc.setZoom(finalScale);
      cancelPendingPaintCommit();
      debugMark('fabric_renderStart', { page: pageNumber, scale: finalScale });
      fc.renderAll();
      debugMark('fabric_renderEnd', { page: pageNumber, scale: finalScale });
      // REMOVED: onScaleApplied callback (old lines 7914-7916)
      schedulePaintCommitted(finalScale);
    };

    if (isCenterPage) {
      // Tier 1: Center page -- immediate render via stagger queue
      enqueueZoomResize(doFabricRender);
    } else if (isVisible) {
      // Tier 2: Visible non-center -- delayed render
      deferredZoomScaleRef.current = { tw, th, finalScale };
      // REMOVED: onScaleApplied callback (old lines 7927-7929)
      const deferTimerId = setTimeout(() => {
        if (inZoomModeRef.current) return;
        if (!deferredZoomScaleRef.current) return;
        enqueueZoomResize(doFabricRender);
      }, 800);
      if (!deferredZoomScaleRef.current) deferredZoomScaleRef.current = { tw, th, finalScale };
      deferredZoomScaleRef.current._timerId = deferTimerId;
    } else {
      // Tier 3: Off-screen -- render when scrolled into view
      deferredZoomScaleRef.current = { tw, th, finalScale };
      // REMOVED: onScaleApplied callback (old lines 7941-7943)
      if (viewportObserverRef.current) viewportObserverRef.current.disconnect();
      viewportObserverRef.current = new IntersectionObserver((entries) => {
        if (!entries[0]?.isIntersecting) return;
        viewportObserverRef.current.disconnect();
        viewportObserverRef.current = null;
        if (!deferredZoomScaleRef.current) return;
        enqueueZoomResize(doFabricRender);
      }, { threshold: 0 });
      viewportObserverRef.current.observe(wrapperEl);
    }

    return;
  }

  // -- Direct resize path (no zoom in progress) --
  pendingScaleRef.current = null;

  if (viewportObserverRef.current) {
    viewportObserverRef.current.disconnect();
    viewportObserverRef.current = null;
  }
  if (deferredZoomScaleRef.current?._timerId) {
    clearTimeout(deferredZoomScaleRef.current._timerId);
  }
  deferredZoomScaleRef.current = null;

  // REMOVED: zoomSettleTimerRef cleanup (old lines 7973-7976)

  if (scaleUpdateFrameRef.current) {
    cancelAnimationFrame(scaleUpdateFrameRef.current);
    scaleUpdateFrameRef.current = null;
  }

  scaleUpdateFrameRef.current = requestAnimationFrame(() => {
    const targetWidth = Math.floor(width * scale);
    const targetHeight = Math.floor(height * scale);
    const canvasZoom = canvas.getZoom();
    const needsResize = canvas.getWidth() !== targetWidth || canvas.getHeight() !== targetHeight;
    const needsZoom = Math.abs(canvasZoom - scale) > 0.0001;

    if (canvas.wrapperEl) {
      canvas.wrapperEl.style.transform = '';
    }

    if (needsResize) {
      canvas.setWidth(targetWidth);
      canvas.setHeight(targetHeight);
    }

    if (needsZoom) {
      canvas.setZoom(scale);
    }

    if (needsResize || needsZoom) {
      cancelPendingPaintCommit();
      canvas.once('after:render', () => {
        // REMOVED: onScaleApplied callback (old lines 8006-8008)
        schedulePaintCommitted(scale);
      });
      canvas.renderAll();
    } else {
      cancelPendingPaintCommit();
      // REMOVED: onScaleApplied callback (old lines 8014-8016)
    }

    scaleUpdateFrameRef.current = null;
  });

  return () => {
    if (scaleUpdateFrameRef.current) {
      cancelAnimationFrame(scaleUpdateFrameRef.current);
      scaleUpdateFrameRef.current = null;
    }
  };
}, [cancelPendingPaintCommit, scale, schedulePaintCommitted, width, height, isInteracting, isZooming, pageNumber]);
// REMOVED from deps: isHidden, onScaleApplied
```
