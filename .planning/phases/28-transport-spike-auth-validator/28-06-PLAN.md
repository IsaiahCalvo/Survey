        }

    ```jsx
    /**
     * Read-only mode gate. Mounted as a sibling of StorageFailureBanner inside YDocProvider.
     * Renders null — all effects are body-attribute + window keydown listener.
     */
    export function ReadOnlyGate() {
      const { accessRevoked } = useYDoc();

      useEffect(() => {
        if (!accessRevoked) {
          // UX: defensive cleanup in case body[data-readonly] was set by a prior mount
          if (typeof document !== 'undefined') {
            document.body?.removeAttribute('data-readonly');
          }
          return undefined;
        }

        // 1. Set body[data-readonly] — CSS in ReadOnlyGate.css dims the existing toolbar
        if (typeof document !== 'undefined') {
          document.body?.setAttribute('data-readonly', 'true');
        }

        // 2. Window-capture-phase keydown listener — suppresses mutation keystrokes
        const onKeyDown = (e) => {
          // UX: Cmd+S MUST pass through (commit 477fe90e UX patch — save is not a mutation)
          if ((e.metaKey || e.ctrlKey) && (e.key === 's' || e.key === 'S')) {
            return;
          }

          // UX: Cmd+Z / Cmd+Shift+Z (undo/redo) suppressed in read-only mode
          if ((e.metaKey || e.ctrlKey) && (e.key === 'z' || e.key === 'Z')) {
            e.preventDefault();
            e.stopPropagation();
            return;
          }

          // UX: Delete / Backspace (annotation deletion) suppressed
          if (e.key === 'Delete' || e.key === 'Backspace') {
            // UX exception: allow Backspace inside form inputs (the ReSignInModal is mounted
            // alongside; users must be able to type a password). Check if the event target
            // is an input/textarea/contenteditable — if so, pass through.
            const t = e.target;
            const tag = t?.tagName?.toLowerCase?.();
            const isEditable =
              tag === 'input' ||
              tag === 'textarea' ||
              t?.isContentEditable === true;
            if (isEditable) return;
            e.preventDefault();
            e.stopPropagation();
            return;
          }

          // (Future: extend with drawing-tool keystrokes if those are wired at window level.
          // Currently the canvas drawing tools attach handlers to canvas nodes themselves,
          // and pointer-events:none on the toolbar already blocks tool selection, so the
          // keyboard list above covers the v2.4 mutation surface.)
        };

        // Capture phase (third arg = true) so we run BEFORE bubble-phase handlers in
        // PageAnnotationLayer or any descendant — this is how we suppress without
        // editing the Always-Protected canvas/SVG components.
        window.addEventListener('keydown', onKeyDown, true);

        // 3. Best-effort active-tool clear via custom event (App.jsx may listen in a
        // future phase; Phase 28 doesn't require the listener)
        try {
          window.dispatchEvent(new CustomEvent('phase28:readonly-activated'));
        } catch { /* swallow — older browsers without CustomEvent constructor */ }

        return () => {
          window.removeEventListener('keydown', onKeyDown, true);
          if (typeof document !== 'undefined') {
            document.body?.removeAttribute('data-readonly');
          }
        };
      }, [accessRevoked]);

      return null;
    }

    export default ReadOnlyGate;
    ```
