    ```js
    let memoizedDeviceId = null;

    function generateUUIDv4() {
      // crypto.randomUUID() preferred (modern browsers + Node 19+), fallback to Math.random for ancient browsers
      if (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function') {
        return crypto.randomUUID();
      }
      // RFC 4122 v4 fallback
      return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
        const r = (Math.random() * 16) | 0;
        const v = c === 'x' ? r : (r & 0x3) | 0x8;
        return v.toString(16);
      });
    }

    export function getDeviceId() {
      if (memoizedDeviceId !== null) return memoizedDeviceId;

      // SSR / Node test env
      if (typeof window === 'undefined') {
        memoizedDeviceId = 'unknown-device';
        return memoizedDeviceId;
      }

      // Tier 1: Electron — main-process os.hostname() exposed via preload
      const electronHostname = window.electronAPI?.osHostname?.();
      if (electronHostname && typeof electronHostname === 'string' && electronHostname.length > 0) {
        memoizedDeviceId = electronHostname;
        return memoizedDeviceId;
      }

      // Tier 2 + 3: web — localStorage stable id
      try {
        const stored = window.localStorage?.getItem('device_id');
        if (stored && typeof stored === 'string' && stored.length > 0) {
          memoizedDeviceId = stored;
          return memoizedDeviceId;
        }
        const fresh = generateUUIDv4();
        window.localStorage?.setItem('device_id', fresh);
        memoizedDeviceId = fresh;
        return memoizedDeviceId;
      } catch {
        // localStorage may throw under strict privacy modes — fall through to ephemeral
        memoizedDeviceId = `ephemeral-${generateUUIDv4()}`;
        return memoizedDeviceId;
      }
    }

    // TEST-ONLY: reset memoization between tests
    export function _resetForTest() {
      memoizedDeviceId = null;
    }
    ```

    ```
    // Phase 28 — Critical auth invariant. THE Pitfall 1 defense.
    // Source: .planning/phases/28-transport-spike-auth-validator/28-RESEARCH.md Pattern 4
    ```

    ```js
    /**
     * Wire Supabase auth events to the Realtime client.
     *
     * @param {object} options
     * @param {object} options.supabase - the Supabase client instance (must expose auth + realtime)
     * @param {() => void} [options.onSignedOut] - called when refresh fails / session ends
     * @returns {{ detach: () => void }}
     */
    export function attachAuthSessionBridge({ supabase, onSignedOut }) {
      if (!supabase?.auth?.onAuthStateChange) {
        throw new Error('[authSessionBridge] supabase.auth.onAuthStateChange unavailable');
      }
      if (!supabase?.realtime?.setAuth) {
        throw new Error('[authSessionBridge] supabase.realtime.setAuth unavailable');
      }

      const { data } = supabase.auth.onAuthStateChange((event, session) => {
        if (event === 'TOKEN_REFRESHED' && session?.access_token) {
          // Pitfall 1 defense: realtime client does NOT auto-pick-up the new token.
          // This is the one line that prevents silent channel-freeze at JWT expiry.
          supabase.realtime.setAuth(session.access_token);
        }
        if (event === 'SIGNED_OUT') {
          // Refresh failed (revoked / password changed / refresh token expired).
          // Surface to caller — typically triggers <ReSignInModal> in Plan 28-06.
          onSignedOut?.();
        }
        // INITIAL_SESSION + SIGNED_IN intentionally not handled here:
        // the Realtime client picks up the session.access_token from the supabase client
        // on first channel.subscribe(). Only mid-session refreshes need explicit setAuth.
      });

      return {
        detach() {
          // unsubscribe is on data.subscription per supabase-js v2.x API
          try { data?.subscription?.unsubscribe?.(); } catch { /* swallow */ }
        },
      };
    }
    ```

    The function is intentionally tiny — ~25 LOC of actual code. The point is the contract, not the volume.

    ```js
    export function uint8ArrayToBase64(bytes) {
      // Browser-safe base64 encode of Uint8Array
      let binary = '';
      const len = bytes.byteLength;
      for (let i = 0; i < len; i++) binary += String.fromCharCode(bytes[i]);
      return btoa(binary);
    }

    export function base64ToUint8Array(b64) {
      const binary = atob(b64);
      const len = binary.length;
      const out = new Uint8Array(len);
      for (let i = 0; i < len; i++) out[i] = binary.charCodeAt(i);
      return out;
    }

    export function encodeUpdate(updateBytes) {
      const encoder = encoding.createEncoder();
      encoding.writeVarUint(encoder, MESSAGE_SYNC);
      syncProtocol.writeUpdate(encoder, updateBytes);
      return encoding.toUint8Array(encoder);
    }

    export function encodeSyncStep1(stateVector) {
      const encoder = encoding.createEncoder();
      encoding.writeVarUint(encoder, MESSAGE_SYNC);
      syncProtocol.writeSyncStep1(encoder, stateVector);
      return encoding.toUint8Array(encoder);
    }

    export function decodeAndApply(ydoc, awareness, frameBytes, origin) {
      const decoder = decoding.createDecoder(frameBytes);
      const messageType = decoding.readVarUint(decoder);
      if (messageType === MESSAGE_SYNC) {
        const replyEncoder = encoding.createEncoder();
        syncProtocol.readSyncMessage(decoder, replyEncoder, ydoc, origin);
        if (encoding.length(replyEncoder) > 1) {
          return encoding.toUint8Array(replyEncoder); // caller broadcasts the reply
        }
        return null;
      }
      if (messageType === MESSAGE_AWARENESS && awareness) {
        awarenessProtocol.applyAwarenessUpdate(awareness, decoding.readVarUint8Array(decoder), origin);
        return null;
      }
      return null;
    }
    ```

    Main provider factory:
    ```js
    /**
     * Create a Supabase Realtime Yjs provider for a single document.
     *
     * @param {object} args
     * @param {string} args.documentId
     * @param {Y.Doc} args.ydoc - borrowed from ydocRegistry; this provider must NOT construct it
     * @param {object} args.supabase - supabase client instance
     * @param {object} [args.awareness] - y-protocols Awareness instance (Phase 33 consumer)
     * @param {(reason: string) => void} [args.onUpdateRejected] - kick / RLS-violation callback
     * @param {(state: 'online'|'offline') => void} [args.onTransportState]
     * @param {() => Promise<{userId:string,deviceId:string,sessionId:string}>} [args.getOriginContext]
     * @returns {{ disconnect: () => void, send: (eventName: string, payload: object) => Promise<void>, getChannel: () => any }}
     */
    export function createSupabaseYjsProvider({ documentId, ydoc, supabase, awareness, onUpdateRejected, onTransportState, getOriginContext }) {
      if (!documentId) throw new Error('[SupabaseYjsProvider] documentId required');
      if (!ydoc) throw new Error('[SupabaseYjsProvider] ydoc required (borrow from ydocRegistry)');
      if (!supabase) throw new Error('[SupabaseYjsProvider] supabase client required');

      const channelName = `yjs:${documentId}`;
      let channel = null;
      let detached = false;

      // Local update fan-out — broadcast outbound, guard echo loops
      const onLocalUpdate = (update, origin) => {
        if (detached) return;
        // Echo-loop guards: do not re-broadcast updates that arrived FROM the network
        if (origin === REMOTE_REALTIME_ORIGIN) return;
        // Phase 27 cross-tab updates (BroadcastChannel) — ydocLifecycle owns REMOTE_BC_ORIGIN
        if (origin && typeof origin === 'object' && origin.source === REMOTE_BC_ORIGIN_TAG) return;

        const frameBytes = encodeUpdate(update);
        if (frameBytes.byteLength > SOFT_PAYLOAD_CAP_BYTES) {
          // Soft cap — Pitfall 2. Phase 32 owns compaction. Log + skip is the spike behavior.
          console.warn('[SupabaseYjsProvider] update exceeds soft cap', frameBytes.byteLength, 'bytes — skipping broadcast');
          return;
        }
        channel?.send({
          type: 'broadcast',
          event: 'sync',
          payload: { update: uint8ArrayToBase64(frameBytes), originClientId: ydoc.clientID },
        }).catch((err) => {
          // Realtime send failed — surface as transport_offline. Will retry on reconnect.
          onTransportState?.('offline');
          console.warn('[SupabaseYjsProvider] send failed', err?.message);
        });
      };
      ydoc.on('update', onLocalUpdate);

      // Subscribe to the channel
      channel = supabase.realtime.channel(channelName, {
        config: { broadcast: { ack: false, self: false } },
      });

      channel
        .on('broadcast', { event: 'sync' }, ({ payload }) => {
          if (detached) return;
          if (payload.originClientId === ydoc.clientID) return; // self-echo guard
          try {
            const bytes = base64ToUint8Array(payload.update);
            const reply = decodeAndApply(ydoc, awareness, bytes, REMOTE_REALTIME_ORIGIN);
            if (reply) {
              channel.send({
                type: 'broadcast',
                event: 'sync',
                payload: { update: uint8ArrayToBase64(reply), originClientId: ydoc.clientID },
              });
            }
          } catch (err) {
            console.warn('[SupabaseYjsProvider] decode failed', err?.message);
          }
        })
        .on('broadcast', { event: 'awareness' }, ({ payload }) => {
          if (detached || !awareness) return;
          try {
            const bytes = base64ToUint8Array(payload.awareness);
            awarenessProtocol.applyAwarenessUpdate(awareness, bytes, REMOTE_REALTIME_ORIGIN);
          } catch (err) {
            console.warn('[SupabaseYjsProvider] awareness decode failed', err?.message);
          }
        })
        .on('broadcast', { event: 'sync_request' }, ({ payload }) => {
          if (detached) return;
          if (payload.fromClientId === ydoc.clientID) return;
          try {
            const remoteSV = base64ToUint8Array(payload.stateVector);
            const reply = encoding.createEncoder();
            encoding.writeVarUint(reply, MESSAGE_SYNC);
            syncProtocol.writeSyncStep2(reply, remoteSV, ydoc);
            const replyBytes = encoding.toUint8Array(reply);
            if (replyBytes.byteLength > 1) {
              channel.send({
                type: 'broadcast',
                event: 'sync',
                payload: { update: uint8ArrayToBase64(replyBytes), originClientId: ydoc.clientID },
              });
            }
          } catch (err) {
            console.warn('[SupabaseYjsProvider] sync_request reply failed', err?.message);
          }
        })
        .subscribe(async (status) => {
          if (detached) return;
          if (status === 'SUBSCRIBED') {
            onTransportState?.('online');
            // Send syncStep1 to ask peers + server snapshot for any updates we don't have
            const stateVector = Y.encodeStateVector(ydoc);
            await channel.send({
              type: 'broadcast',
              event: 'sync_request',
              payload: { stateVector: uint8ArrayToBase64(stateVector), fromClientId: ydoc.clientID },
            });
          }
          if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT' || status === 'CLOSED') {
            onTransportState?.('offline');
          }
        });

      return {
        disconnect() {
          detached = true;
          try { ydoc.off('update', onLocalUpdate); } catch { /* swallow */ }
          try { channel?.unsubscribe(); } catch { /* swallow */ }
          try { supabase.realtime.removeChannel(channel); } catch { /* swallow */ }
          channel = null;
        },
        send(eventName, payload) {
          if (detached || !channel) return Promise.resolve();
          return channel.send({ type: 'broadcast', event: eventName, payload });
        },
        getChannel() { return channel; },
      };
    }
    ```
