        fontColor: '#1e293b',
        borderOpacity: 1,
        fillOpacity: 1,
      },

    // Helper: walk a React element tree looking for children with a matching
    // data-* prop. React elements expose .props.children which is either a
    // single element, an array, or a string.
    function findAll(el, predicate, found = []) {
      if (!el || typeof el !== 'object') return found;
      if (predicate(el)) found.push(el);
      const children = el.props && el.props.children;
      if (Array.isArray(children)) {
        children.forEach((c) => findAll(c, predicate, found));
      } else if (children) {
        findAll(children, predicate, found);
      }
      return found;
    }

    test('renderCallout returns a React element for a valid callout', () => {
      const out = renderCallout(baseCallout, 0, { width: 1000, height: 800 }, stubConnection);
      assert.ok(out, 'expected non-null render output');
      assert.equal(out.type, 'g', 'outer element should be <g>');

    test('renderCallout outer <g> carries data-callout-id', () => {

    function isUserTyping(activeElement) {
      if (!activeElement) return false;
      if (activeElement.tagName === 'INPUT') return true;
      if (activeElement.tagName === 'TEXTAREA') return true;
      if (activeElement.isContentEditable === true) return true;
      if (activeElement.contentEditable === 'true') return true;
      if (typeof activeElement.closest === 'function' &&
          activeElement.closest('.fabric-hidden-textarea')) return true;
      return false;
    }

    // Pure-logic delete dispatch — returns which callback was called, for
    // assertions. Plan 14-02 inlines this in SVGAnnotationLayer.jsx effect.
    function dispatchDelete({
      key,
      isTyping,
      selectedIds,
      selectedCalloutIds,
      deleteSelected,
      onDeleteSelectedCallouts,
    }) {
      if (key !== 'Delete' && key !== 'Backspace') return null;
      if (isTyping) return null;
      if (selectedIds && selectedIds.length > 0) {
        deleteSelected();
        return 'annotations';
      }
      if (selectedCalloutIds && selectedCalloutIds.length > 0) {
        onDeleteSelectedCallouts(selectedCalloutIds);
        return 'callouts';
      }
      return null;
    }

      assert.equal(isUserTyping({ tagName: 'INPUT' }), true);
    });

    **Step 1: Create `src/utils/calloutEditAdapter.js`** (new file, ~120-150 LOC)

    This is a PURE utility module. It imports `fabric` from `fabric` at module top (standard pattern). It exports THREE functions: `sanitizeFontFamily`, `toFabricGroup`, `fromFabricGroup`.

    ```javascript
    // src/utils/calloutEditAdapter.js
    // Adapter between React callout shape (normalized coords) and Fabric.js Group
    // shape (page coords) that loadCalloutAnnotation in FabricEditCanvas.jsx:1937
    // already expects. Pure utility — no React, no component imports.

    import { fabric } from 'fabric';

    /**
     * Strip CSS fallback stacks down to a single font name.
     * Required because Fabric.js Textbox measures chars at CACHE_FONT_SIZE=400px
     * and the browser may resolve different fonts in a fallback chain at 400px vs
     * actual display size, producing wrong widths and cursor drift.
     * See 14-RESEARCH.md Pitfall 2 (cursor drift bug 2026-04-08).
     */
    export function sanitizeFontFamily(raw) {
      if (raw == null) return 'Arial';
      const first = String(raw).split(',')[0];
      if (!first) return 'Arial';
      const stripped = first.trim().replace(/^['"]|['"]$/g, '');
      return stripped.length > 0 ? stripped : 'Arial';
    }

    /**
     * Convert a React callout (normalized 0-1 coords) into a Fabric.js Group
     * with page-coordinate children that loadCalloutAnnotation can enliven.
     *
     * Input:  reactCallout (see types.js createCallout shape), pageSize { width, height }
     * Output: fabric.Group with `reactCalloutId` stashed for commit routing
     */
    export function toFabricGroup(reactCallout, pageSize) {
      const { width: W, height: H } = pageSize;
      const at = reactCallout.arrowTip;
      const k = reactCallout.knee;
      const tb = reactCallout.textBoxPosition;
      const tbW = reactCallout.textBoxWidth * W;
      const tbH = reactCallout.textBoxHeight * H;
      const tbX = tb.x * W;
      const tbY = tb.y * H;
      const style = reactCallout.style || {};

      const stroke = style.borderColor || '#1e293b';
      const strokeWidth = style.lineThickness || 2;

      const line1 = new fabric.Line(
        [tbX + tbW / 2, tbY + tbH / 2, k.x * W, k.y * H],
        { stroke, strokeWidth, strokeUniform: true }
      );
      const line2 = new fabric.Line(
        [k.x * W, k.y * H, at.x * W, at.y * H],
        { stroke, strokeWidth, strokeUniform: true }
      );
      const tipDot = new fabric.Circle({
        left: at.x * W - 3,
        top: at.y * H - 3,
        radius: 3,
        fill: stroke,
      });
      const rect = new fabric.Rect({
        left: tbX,
        top: tbY,
        width: tbW,
        height: tbH,
        fill: style.fillColor || '#ffffff',
        stroke,
        strokeWidth,
        rx: 4,
        ry: 4,
        strokeUniform: true,
      });
      const textbox = new fabric.Textbox(reactCallout.text || '', {
        left: tbX + 4,
        top: tbY + 4,
        width: Math.max(8, tbW - 8),
        fontSize: style.fontSize || 14,
        // Pitfall 2: single-name fontFamily only — strip fallback stacks
        fontFamily: sanitizeFontFamily(style.fontFamily),
        fontWeight: style.bold ? 'bold' : 'normal',
        fill: style.fontColor || '#1e293b',
      });

      const group = new fabric.Group([line1, line2, rect, tipDot, textbox], {
        // FabricEditCanvas.jsx:1937 loadCalloutAnnotation identifies callout
        // groups via data: { type: 'callout' } on the group. Mirror that.
        data: { type: 'callout' },
      });
      // Plan 14-03's App.jsx commit wrapper uses this field to route the save
      // back to setCallouts instead of setAnnotationsByPage.
      group.reactCalloutId = reactCallout.id;
      group.set('reactCalloutId', reactCallout.id);
      return group;
    }

    /**
     * Reverse adapter: convert an edited Fabric Group back to the React callout
     * shape (normalized 0-1 coords). Plan 14-03 calls this inside the App.jsx
     * save-callback wrapper after loadCalloutAnnotation commits the Fabric edit.
     *
     * NOTE: For the N=10 round-trip precision requirement, this function must
     * be arithmetically exact — no intermediate scaling that rounds. The test
     * tolerance is 1e-6, but the pure math is integer-clean so the result
     * should match to float precision if you avoid accumulation errors.
     */
    export function fromFabricGroup(fabricGroup, pageSize, originalReactCallout) {
      const { width: W, height: H } = pageSize;
      const children = typeof fabricGroup.getObjects === 'function'
        ? fabricGroup.getObjects()
        : (fabricGroup._objects || []);

      // Identify children by index since the toFabricGroup ordering is fixed:
      // [line1, line2, rect, tipDot, textbox]
      const line1 = children[0];
      const line2 = children[1];
      const rect = children[2];
      const tipDot = children[3];
      const textbox = children[4];

      // Extract updated page-coord positions and normalize.
      // Line1 runs from textbox-center to knee; use line1 endpoints to locate
      // the knee. Line2 runs from knee to arrowTip.
      const kneePx = { x: line1?.x2 ?? 0, y: line1?.y2 ?? 0 };
      const arrowTipPx = { x: line2?.x2 ?? 0, y: line2?.y2 ?? 0 };
      const rectLeft = rect?.left ?? 0;
      const rectTop = rect?.top ?? 0;
      const rectWidth = (rect?.width ?? 0) * (rect?.scaleX ?? 1);
      const rectHeight = (rect?.height ?? 0) * (rect?.scaleY ?? 1);

      return {
        ...originalReactCallout,
        arrowTip: { x: arrowTipPx.x / W, y: arrowTipPx.y / H },
        knee: { x: kneePx.x / W, y: kneePx.y / H },
        textBoxPosition: { x: rectLeft / W, y: rectTop / H },
        textBoxWidth: rectWidth / W,
        textBoxHeight: rectHeight / H,
        text: textbox?.text ?? originalReactCallout.text ?? '',
      };
    }
    ```

    **Precision note:** The round-trip test (tests/calloutEditAdapter.test.mjs Test 6) asserts 10-cycle drift < 1e-6. Do NOT add scale multipliers or pre-emptive offsets. Keep the math integer-clean.
